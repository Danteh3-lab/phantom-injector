cmake_minimum_required(VERSION 3.20)
project(PhantomDumper VERSION 0.6.0 LANGUAGES CXX)

include(CTest)
add_library(PhantomDumperCore STATIC src/Logger.cpp src/Runtime.cpp src/MemoryInspector.cpp
    src/PatternScanner.cpp src/StructureInspector.cpp src/OffsetManager.cpp src/ExportManager.cpp)
target_include_directories(PhantomDumperCore PUBLIC include)
target_compile_features(PhantomDumperCore PUBLIC cxx_std_17)
set_target_properties(PhantomDumperCore PROPERTIES POSITION_INDEPENDENT_CODE ON)

if(WIN32)
    if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "PhantomDumper requires a 64-bit Windows build")
    endif()
    target_sources(PhantomDumperCore PRIVATE src/WindowsMemoryBackend.cpp)
    target_compile_definitions(PhantomDumperCore PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    add_library(PhantomDumper SHARED src/dllmain.cpp src/Exports.cpp src/ModuleResolver.cpp)
    target_link_libraries(PhantomDumper PRIVATE PhantomDumperCore)
    target_compile_definitions(PhantomDumper PRIVATE PHANTOM_DUMPER_BUILD UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    set_target_properties(PhantomDumper PROPERTIES PREFIX "")
endif()

if(BUILD_TESTING)
    find_package(Threads REQUIRED)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(PhantomDumperCoreTests tests/CoreTests.cpp)
    target_link_libraries(PhantomDumperCoreTests PRIVATE PhantomDumperCore Threads::Threads)
    add_test(NAME PhantomDumper.Core COMMAND PhantomDumperCoreTests)
    add_executable(PhantomDumperMemoryTests tests/MemoryInspectorTests.cpp)
    target_link_libraries(PhantomDumperMemoryTests PRIVATE PhantomDumperCore)
    add_test(NAME PhantomDumper.Memory COMMAND PhantomDumperMemoryTests)
    add_executable(PhantomDumperPatternTests tests/PatternScannerTests.cpp)
    target_link_libraries(PhantomDumperPatternTests PRIVATE PhantomDumperCore)
    add_test(NAME PhantomDumper.Patterns COMMAND PhantomDumperPatternTests)
    add_executable(PhantomDumperStructureTests tests/StructureInspectorTests.cpp)
    target_link_libraries(PhantomDumperStructureTests PRIVATE PhantomDumperCore)
    add_test(NAME PhantomDumper.Structures COMMAND PhantomDumperStructureTests)
    add_executable(PhantomDumperOffsetTests tests/OffsetManagerTests.cpp)
    target_link_libraries(PhantomDumperOffsetTests PRIVATE PhantomDumperCore)
    add_test(NAME PhantomDumper.Offsets COMMAND PhantomDumperOffsetTests)
    add_executable(PhantomDumperExportTests tests/ExportManagerTests.cpp)
    target_link_libraries(PhantomDumperExportTests PRIVATE PhantomDumperCore)
    add_test(NAME PhantomDumper.Export COMMAND PhantomDumperExportTests)
    add_executable(PhantomDumperExportFixture tests/ExportFixture.cpp)
    target_link_libraries(PhantomDumperExportFixture PRIVATE PhantomDumperCore)
    if(WIN32)
        target_sources(PhantomDumperExportFixture PRIVATE src/ModuleResolver.cpp)
        target_compile_definitions(PhantomDumperExportFixture PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    endif()
    set(export_fixture_directory "${CMAKE_CURRENT_BINARY_DIR}/export-fixtures/$<CONFIG>")
    set(export_fixture_outputs "${export_fixture_directory}/fixture.hpp" "${export_fixture_directory}/empty.hpp"
        "${export_fixture_directory}/fixture.json" "${export_fixture_directory}/empty.json")
    add_custom_command(OUTPUT ${export_fixture_outputs}
        COMMAND $<TARGET_FILE:PhantomDumperExportFixture> "${export_fixture_directory}"
        DEPENDS PhantomDumperExportFixture VERBATIM)
    add_custom_target(PhantomDumperExportFixtures DEPENDS ${export_fixture_outputs})
    add_executable(PhantomDumperGeneratedHeaderTests tests/GeneratedHeaderTests.cpp)
    add_executable(PhantomDumperEmptyHeaderTests tests/EmptyHeaderTests.cpp)
    foreach(target IN ITEMS PhantomDumperGeneratedHeaderTests PhantomDumperEmptyHeaderTests)
        add_dependencies(${target} PhantomDumperExportFixtures)
        target_include_directories(${target} PRIVATE "${export_fixture_directory}")
        target_compile_features(${target} PRIVATE cxx_std_17)
    endforeach()
    add_test(NAME PhantomDumper.ExportWorkflow COMMAND PhantomDumperExportFixture "${export_fixture_directory}")
    set_tests_properties(PhantomDumper.ExportWorkflow PROPERTIES FIXTURES_SETUP export_outputs)
    add_test(NAME PhantomDumper.ExportJson COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/tests/VerifyExportJson.py" "${export_fixture_directory}")
    set_tests_properties(PhantomDumper.ExportJson PROPERTIES FIXTURES_REQUIRED export_outputs)
    add_test(NAME PhantomDumper.GeneratedHeader COMMAND PhantomDumperGeneratedHeaderTests)
    add_test(NAME PhantomDumper.EmptyHeader COMMAND PhantomDumperEmptyHeaderTests)
    if(WIN32)
        add_executable(PhantomDumperWindowsMemoryTests tests/WindowsMemoryTests.cpp)
        target_sources(PhantomDumperWindowsMemoryTests PRIVATE src/ModuleResolver.cpp)
        target_link_libraries(PhantomDumperWindowsMemoryTests PRIVATE PhantomDumperCore)
        target_compile_definitions(PhantomDumperWindowsMemoryTests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
        add_test(NAME PhantomDumper.WindowsMemory COMMAND PhantomDumperWindowsMemoryTests)
        add_executable(PhantomDumperSmoke tests/SmokeHost.cpp)
        target_compile_features(PhantomDumperSmoke PRIVATE cxx_std_17)
        target_compile_definitions(PhantomDumperSmoke PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
        add_test(NAME PhantomDumper.WindowsLifecycle COMMAND PhantomDumperSmoke $<TARGET_FILE:PhantomDumper>)
    endif()
endif()

get_property(dumper_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
foreach(target IN LISTS dumper_targets)
    get_target_property(target_type ${target} TYPE)
    if(target_type STREQUAL "UTILITY")
        continue()
    endif()
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /EHsc /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()
